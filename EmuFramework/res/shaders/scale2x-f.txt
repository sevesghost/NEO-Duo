 * Original implementation of the published 2x pixel rule:
 * the output of a quadrant is the adjacent neighbor when that neighbor
 * matches the other adjacent neighbor and both differ from the far pair.
 */
in lowp vec2 texUVOut[5];
in mediump vec2 texelPosOut;

void main()
{
	precision lowp float;

	vec4 center = TEXTURE(TEX, texUVOut[2]);
	vec4 west = TEXTURE(TEX, texUVOut[1]);
	vec4 east = TEXTURE(TEX, texUVOut[3]);
	vec4 north = TEXTURE(TEX, texUVOut[0]);
	vec4 south = TEXTURE(TEX, texUVOut[4]);
	vec2 quadrant = fract(texelPosOut);
	vec4 across = quadrant.x >= 0.5 ? east : west;
	vec4 along = quadrant.y >= 0.5 ? south : north;
	vec4 oppositeAcross = quadrant.x >= 0.5 ? west : east;
	vec4 oppositeAlong = quadrant.y >= 0.5 ? north : south;
	if(across == along && along != oppositeAlong && across != oppositeAcross)
		center = across;
	FRAGCOLOR = center;
}
