 * Original neighborhood for the 2x rule below. Outputs are the center sample
 * and its four orthogonal neighbors, plus the pixel position in source texels.
 */
uniform vec2 srcTexelDelta;
uniform vec2 srcPixels;
in vec2 texUV;
out vec2 texUVOut[5];
out vec2 texelPosOut;

void main()
{
	vec2 stepX = vec2(srcTexelDelta.x, 0.0);
	vec2 stepY = vec2(0.0, srcTexelDelta.y);
	texUVOut[0] = texUV - stepY;
	texUVOut[1] = texUV - stepX;
	texUVOut[2] = texUV;
	texUVOut[3] = texUV + stepX;
	texUVOut[4] = texUV + stepY;
	texelPosOut = texUV * srcPixels;
	gl_Position = POS;
}
