 * Original edge-directed smoother. It blends the center sample toward the
 * axis whose neighbors differ less.
 */
in lowp vec2 texUVOut[9];

void main()
{
	#ifdef GL_FRAGMENT_PRECISION_HIGH
	precision highp float;
	#else
	precision mediump float;
	#endif

	vec3 northWest = TEXTURE(TEX, texUVOut[0]).rgb;
	vec3 north = TEXTURE(TEX, texUVOut[1]).rgb;
	vec3 northEast = TEXTURE(TEX, texUVOut[2]).rgb;
	vec3 west = TEXTURE(TEX, texUVOut[3]).rgb;
	vec3 center = TEXTURE(TEX, texUVOut[4]).rgb;
	vec3 east = TEXTURE(TEX, texUVOut[5]).rgb;
	vec3 southWest = TEXTURE(TEX, texUVOut[6]).rgb;
	vec3 south = TEXTURE(TEX, texUVOut[7]).rgb;
	vec3 southEast = TEXTURE(TEX, texUVOut[8]).rgb;
	const vec3 lumaWeight = vec3(0.299, 0.587, 0.114);
	float horizontal = abs(dot(west - east, lumaWeight));
	float vertical = abs(dot(north - south, lumaWeight));
	float diagonalA = abs(dot(northWest - southEast, lumaWeight));
	float diagonalB = abs(dot(northEast - southWest, lumaWeight));
	vec3 along;
	float spread;
	if(horizontal + diagonalA < vertical + diagonalB)
	{
		along = (west + east) * 0.5;
		spread = vertical - horizontal;
	}
	else
	{
		along = (north + south) * 0.5;
		spread = horizontal - vertical;
	}
	float amount = clamp(spread * 4.0, 0.0, 0.65);
	FRAGCOLOR = vec4(mix(center, along, amount), 1.0);
}
