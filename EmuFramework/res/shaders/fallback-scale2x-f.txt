 * Original implementation of the published 2x pixel rule for one varying.
 * A quadrant copies the adjacent neighbor when that neighbor matches the
 * other adjacent neighbor and both differ from the far pair.
 */
uniform lowp vec2 srcTexelDelta;
in lowp vec2 texUVOut;
in mediump vec2 texelPosOut;

void main()
{
	precision lowp float;

	vec2 stepX = vec2(srcTexelDelta.x, 0.0);
	vec2 stepY = vec2(0.0, srcTexelDelta.y);
	vec4 center = TEXTURE(TEX, texUVOut);
	vec4 west = TEXTURE(TEX, texUVOut - stepX);
	vec4 east = TEXTURE(TEX, texUVOut + stepX);
	vec4 north = TEXTURE(TEX, texUVOut - stepY);
	vec4 south = TEXTURE(TEX, texUVOut + stepY);
	vec2 quadrant = fract(texelPosOut);
	vec4 across = quadrant.x >= 0.5 ? east : west;
	vec4 along = quadrant.y >= 0.5 ? south : north;
	vec4 oppositeAcross = quadrant.x >= 0.5 ? west : east;
	vec4 oppositeAlong = quadrant.y >= 0.5 ? north : south;
	if(across == along && along != oppositeAlong && across != oppositeAcross)
		center = across;
	FRAGCOLOR = center;
}
