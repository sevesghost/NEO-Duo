 * Original edge-directed smoother for targets that cannot pass nine varyings.
 * The fragment gathers the 3x3 neighborhood from one coordinate.
 */
uniform lowp vec2 srcTexelHalfDelta;
in lowp vec2 texUVOut;

void main()
{
	#ifdef GL_FRAGMENT_PRECISION_HIGH
	precision highp float;
	#else
	precision mediump float;
	#endif

	vec2 stepX = vec2(srcTexelHalfDelta.x, 0.0);
	vec2 stepY = vec2(0.0, srcTexelHalfDelta.y);
	vec3 north = TEXTURE(TEX, texUVOut - stepY).rgb;
	vec3 south = TEXTURE(TEX, texUVOut + stepY).rgb;
	vec3 west = TEXTURE(TEX, texUVOut - stepX).rgb;
	vec3 east = TEXTURE(TEX, texUVOut + stepX).rgb;
	vec3 center = TEXTURE(TEX, texUVOut).rgb;
	const vec3 lumaWeight = vec3(0.299, 0.587, 0.114);
	float horizontal = abs(dot(west - east, lumaWeight));
	float vertical = abs(dot(north - south, lumaWeight));
	vec3 along = horizontal < vertical ? (west + east) * 0.5 : (north + south) * 0.5;
	float amount = clamp(abs(horizontal - vertical) * 4.0, 0.0, 0.65);
	FRAGCOLOR = vec4(mix(center, along, amount), 1.0);
}
